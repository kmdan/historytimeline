def dist(lon0,lat0,sy=1.0):
  p1,p2=np.radians(LAT),np.radians(lat0); dl=np.radians(LON-lon0)
  base=6371*np.arccos(np.clip(np.sin(p1)*np.sin(p2)+np.cos(p1)*np.cos(p2)*np.cos(dl),-1,1))
  if sy!=1.0:
    dy=(LAT-lat0)*111.2; dx=np.sqrt(np.maximum(base**2-dy**2,0)); return np.sqrt(dx**2+(dy/sy)**2)
  return base
