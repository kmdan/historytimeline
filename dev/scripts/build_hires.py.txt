import numpy as np, gzip, json
from PIL import Image, ImageFilter
from scipy import ndimage
Image.MAX_IMAGE_PIXELS=None
GW,GH=2160,1080
BM='bmx/mpl_toolkits/basemap_data/'
# --- elevation (ETOPO20 upsampled)
d=np.load('etopo.npy')[:, :-1]; lo=np.load('lo.npy')[:-1]
lo=np.where(lo>180,lo-360,lo); idx=np.argsort(lo); d=d[:,idx][::-1]
d2=ndimage.zoom(d,2,order=1,mode='nearest')[:GH,:GW]
# --- modern land/lake mask (5 arc-min, rows from south)
m=np.frombuffer(gzip.decompress(open(BM+'lsmask_5min_h.bin','rb').read()),np.uint8).reshape(2160,4320)[::-1]
mb=m.reshape(GH,2,GW,2)
land=(mb==1).sum((1,3))>=2; lake=((mb==2).sum((1,3))>=2)&~land
R=np.zeros((GH,GW),np.uint8)
e=np.maximum(d2,1); R[land]=(140+np.minimum(115,np.round(np.sqrt(e[land])/np.sqrt(6000)*115))).astype(np.uint8)
oc=~land&~lake; dep=np.maximum(-d2,1)
R[oc&(dep<=130)]=np.round(dep[oc&(dep<=130)]).clip(1,130).astype(np.uint8)
deep=oc&(dep>130); R[deep]=(131+np.minimum(8,np.floor(np.log2(dep[deep]/130)*2))).astype(np.uint8)
R[lake]=135
# --- relief detail from Natural Earth shaded relief
sr=Image.open(BM+'shadedrelief.jpg').convert('L').resize((GW,GH),Image.LANCZOS)
L=np.asarray(sr).astype(float)+8
bl=ndimage.gaussian_filter(L,5)
G=np.clip(128+(L/bl-1)*420,0,255).astype(np.uint8)
# --- noise (same field upsampled)
NZ=np.load('/tmp/claude-0/NZ.npy'); NZ2=ndimage.zoom(NZ,2,order=3)[:GH,:GW]
fine=np.random.default_rng(5).normal(size=(GH//4+2,GW//4+2)); fine=ndimage.zoom(fine,4,order=3)[:GH,:GW]
NZ2=np.clip(NZ2+fine*0.04,0,1)
B=(NZ2*255).astype(np.uint8)
Image.fromarray(np.dstack([R,G,B]),'RGB').save('/home/claude/atlas/terrain.png',optimize=True)
np.save('/tmp/claude-0/NZ2.npy',NZ2); np.save('/tmp/claude-0/land2.npy',land); np.save('/tmp/claude-0/d2.npy',d2)
# --- ice & farm: upsample code fields
ice=np.array(Image.open('/home/claude/atlas/ice.png')).astype(float)
on=np.where(ice[...,0]==0,255,ice[...,0]); on2=ndimage.zoom(on,2,order=1)[:GH,:GW]; on2=np.where(on2>=254.5,0,np.round(on2))
off=ice[...,1]; off2=ndimage.zoom(off,2,order=1)[:GH,:GW]; off2=np.where(off2>=254.5,255,np.minimum(254,np.round(off2)))
fa=np.where(ice[...,2]==0,255,ice[...,2]); fa2=ndimage.zoom(fa,2,order=1)[:GH,:GW]; fa2=np.where(fa2>=254.5,0,np.round(fa2))
fa2[~land]=0
# ice only matters on land, exposed shelf or lakes; keep as is
Image.fromarray(np.dstack([on2,off2,fa2]).astype(np.uint8),'RGB').save('/home/claude/atlas/ice.png',optimize=True)
print('terrain+ice ok')
